      const s = (ev.startTime || "").slice(0, 5);
      const e = (ev.endTime || "").slice(0, 5);
      if (ev.startDate === ev.endDate) {
        const t = (s || e) ? ` ${s}${e ? "~" + e : "~"}` : "";
        return `${fmtShort(ev.startDate)}${t}`;
      }
      if (ev.kind === "outing") {
        return `${fmtShort(ev.startDate)}${s ? " " + s : ""} ~ ${fmtShort(ev.endDate)}${e ? " " + e : ""}`;
      }
      const t = (s || e) ? ` · ${s}${e ? "~" + e : "~"}` : "";
      return `${fmtShort(ev.startDate)} ~ ${fmtShort(ev.endDate)}${t}`;
    }
    function fmtDayTimeBadge(ev, ymd) {
      const s = (ev.startTime || "").slice(0, 5);
      const e = (ev.endTime || "").slice(0, 5);
      if (!s && !e) return "";
      if (ev.kind === "outing") {
        if (ev.startDate === ev.endDate) {
          return s && e ? `${s}~${e}` : (s ? `출타 ${s}` : `복귀 ${e}`);
        }
        const parts = [];
        if (ev.startDate === ymd && s) parts.push(`출타 ${s}`);
        if (ev.endDate === ymd && e) parts.push(`복귀 ${e}`);
        return parts.join(" · ");
      }
      if (s && e) return `${s}~${e}`;
      return s ? `${s}~` : `~${e}`;
    }
    function fmtRelative(isoOrMs) {
      if (!isoOrMs) return "방금";
      const t = typeof isoOrMs === "number" ? isoOrMs : new Date(isoOrMs).getTime();
      const diff = Math.max(0, Math.floor((Date.now() - t) / 1000));
      if (diff < 60) return "방금 전";
      if (diff < 3600) return `${Math.floor(diff / 60)}분 전`;
      if (diff < 86400) return `${Math.floor(diff / 3600)}시간 전`;
      if (diff < 86400 * 7) return `${Math.floor(diff / 86400)}일 전`;
      const d = new Date(t);
      return `${d.getMonth() + 1}/${d.getDate()}`;
    }
    function fmtRemainMins(mins) {
      if (mins <= 0) return "곧";
      const h = Math.floor(mins / 60);
      const m = mins % 60;
      if (h > 0 && m > 0) return `${h}시간 ${m}분`;
      if (h > 0) return `${h}시간`;
      return `${m}분`;
    }
    function esc(str) {
      return String(str ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    function resolveMonthDay(month, day, baseDate = new Date()) {
      let year = baseDate.getFullYear();
      const curM = baseDate.getMonth() + 1;
      if (month < curM - 5) year += 1;
      return `${year}-${pad2(month)}-${pad2(day)}`;
    }


    function to24h(h, meridiem) {
      let hr = Number(h);
      if (meridiem === "오후" || meridiem === "저녁" || meridiem === "밤") {
        if (hr < 12) hr += 12;
      } else if (meridiem === "오전" || meridiem === "새벽") {
        if (hr === 12) hr = 0;
      }
      return Math.max(0, Math.min(23, hr));
    }


    function parseQuickInput(rawInput, fallbackDate) {
      let text = String(rawInput || "").trim();
      if (!text) return null;


      const out = {};
      const today = todayYMD();
      const now = new Date();


      // 1. Optional [이름] prefix
      const authorM = text.match(/^\[([^\]]{1,12})\]\s*/);
      if (authorM) {
        const tag = authorM[1].trim();
        if (!["출타","휴가","공유","가안","확정","알바","개인","외박","외출"].includes(tag)) {
          out.author = tag;
        }
        text = text.slice(authorM[0].length);
      }


      // 2. Date Ranges
      let dateMatched = false;
      // 2a. YYYY-MM-DD ~ YYYY-MM-DD
      let m = text.match(/(\d{4})[.\-/](\d{1,2})[.\-/](\d{1,2})\s*(?:~|-|부터)\s*(\d{4})[.\-/](\d{1,2})[.\-/](\d{1,2})(?:까지)?/);
      if (m) {
        out.startDate = `${m[1]}-${pad2(m[2])}-${pad2(m[3])}`;
        out.endDate = `${m[4]}-${pad2(m[5])}-${pad2(m[6])}`;
        text = text.replace(m[0], " ");
        dateMatched = true;
      }
      // 2b. M/D ~ M/D or M/D ~ D
      if (!dateMatched) {
        m = text.match(/(?:^|\s)(\d{1,2})\/(\d{1,2})\s*(?:~|-|부터)\s*(?:(\d{1,2})\/)?(\d{1,2})(?:까지)?(?=\s|$)/);
        if (m) {
          const sm = Number(m[1]), sd = Number(m[2]);
          const em = m[3] ? Number(m[3]) : sm;
          const ed = Number(m[4]);
          out.startDate = resolveMonthDay(sm, sd, now);
          out.endDate = resolveMonthDay(em, ed, parseYMD(out.startDate));
          if (out.endDate < out.startDate) {
            const d = parseYMD(out.endDate);
            d.setFullYear(d.getFullYear() + 1);
            out.endDate = toYMD(d);
          }
          text = text.replace(m[0], " ");
          dateMatched = true;
        }
      }
      // 2c. M월 D일 ~ (M월)? D일
      if (!dateMatched) {
        m = text.match(/(\d{1,2})\s*월\s*(\d{1,2})\s*일(?:부터)?\s*(?:~|-|에서)?\s*(?:(\d{1,2})\s*월\s*)?(\d{1,2})\s*일(?:까지)?/);
        if (m) {
          const sm = Number(m[1]), sd = Number(m[2]);
          const em = m[3] ? Number(m[3]) : sm;
          const ed = Number(m[4]);
          out.startDate = resolveMonthDay(sm, sd, now);
          out.endDate = resolveMonthDay(em, ed, parseYMD(out.startDate));
          text = text.replace(m[0], " ");
          dateMatched = true;
        }
      }


      // 3. Single Dates
      if (!dateMatched) {
        if ((m = text.match(/(\d{4})[.\-/](\d{1,2})[.\-/](\d{1,2})/))) {
          out.startDate = `${m[1]}-${pad2(m[2])}-${pad2(m[3])}`;
          text = text.replace(m[0], " ");
          dateMatched = true;
        } else if ((m = text.match(/(?:^|\s)(\d{1,2})\/(\d{1,2})(?:부터)?(?=\s|$)/))) {
          out.startDate = resolveMonthDay(Number(m[1]), Number(m[2]), now);
          text = text.replace(m[0], " ");
          dateMatched = true;
        } else if ((m = text.match(/(\d{1,2})\s*월\s*(\d{1,2})\s*일(?:부터)?/))) {
          out.startDate = resolveMonthDay(Number(m[1]), Number(m[2]), now);
          text = text.replace(m[0], " ");
          dateMatched = true;
        } else if ((m = text.match(/(?:^|\s)(오늘|내일|모레|글피)(?:부터)?(?=\s|$)/))) {
          const map = { "오늘": 0, "내일": 1, "모레": 2, "글피": 3 };
          out.startDate = addDays(today, map[m[1]]);
          text = text.replace(m[0], " ");
          dateMatched = true;
        } else if ((m = text.match(/(?:^|\s)(?:(이번\s*주|다음\s*주|담\s*주|차주)\s*)?([일월화수목금토])(?:요일|욜)?(?:부터)?(?=\s|$)/))) {
          const wkPrefix = (m[1] || "").replace(/\s+/g, "");
          const targetDow = WEEKDAYS.indexOf(m[2]);
          const curDow = now.getDay();
          let delta = targetDow - curDow;
          if (wkPrefix === "다음주" || wkPrefix === "담주" || wkPrefix === "차주") {
            const daysToNextMon = ((8 - curDow) % 7) || 7;
            const targetFromMon = targetDow === 0 ? 6 : targetDow - 1;
            delta = daysToNextMon + targetFromMon;
          } else if (wkPrefix === "이번주") {
            if (delta < 0) delta += 7;
          } else {
            if (delta < 0) delta += 7;
          }
          out.startDate = addDays(today, delta);
          text = text.replace(m[0], " ");
          dateMatched = true;
        }
      }


      if (!out.startDate) out.startDate = fallbackDate || today;


      // 4. Duration (N박M일 / 당일 / N일간)
      let durationDays = null;
      if ((m = text.match(/(\d+)\s*박\s*(\d+)\s*일(?:간)?/))) {
        durationDays = Math.max(1, Number(m[2]));
        text = text.replace(m[0], " ");
      } else if ((m = text.match(/(\d+)\s*일간/))) {
        durationDays = Math.max(1, Number(m[1]));
        text = text.replace(m[0], " ");
      } else if ((m = text.match(/(?:^|\s)당일(?=\s|$)/))) {
        durationDays = 1;
        text = text.replace(m[0], " ");
      }


      // 5. Repeat Weeks
      if ((m = text.match(/(\d+)\s*주\s*(?:반복|간)?/))) {
        const w = Number(m[1]);
        if ([1, 2, 4, 8].includes(w)) out.repeatWeeks = w;
        else if (w > 1) out.repeatWeeks = w <= 3 ? 2 : (w <= 6 ? 4 : 8);
        text = text.replace(m[0], " ");
      }
      if (/(?:^|\s)매주(?=\s|$)/.test(text)) {
        if (!out.repeatWeeks) out.repeatWeeks = 4;
        text = text.replace(/(?:^|\s)매주(?=\s|$)/g, " ");
      }


      // 6. Time Ranges & Individual Times
      // 6a. Full time range (e.g. 08:30~21:00, 오후 2시~10시, 9시반~21시)
      const timeRangeRe = /(오전|오후|저녁|밤|새벽)?\s*(\d{1,2})\s*(?::(\d{2})|시\s*(\d{1,2}|반)?\s*분?)\s*(?:~|-|부터)\s*(오전|오후|저녁|밤|새벽)?\s*(\d{1,2})\s*(?::(\d{2})|시\s*(\d{1,2}|반)?\s*분?)(?:까지)?/;
      const compactTimeRe = /(오전|오후|저녁|밤|새벽)?\s*(\d{1,2})\s*(?:~|-)\s*(오전|오후|저녁|밤|새벽)?\s*(\d{1,2})\s*시(?:까지)?/;


      if ((m = text.match(timeRangeRe))) {
        const sh = to24h(m[2], m[1]);
        const sm = m[3] ? Number(m[3]) : (m[4] === "반" ? 30 : Number(m[4] || 0));
        let eh = to24h(m[6], m[5] || (m[1] && Number(m[6]) < 12 && Number(m[6]) >= Number(m[2]) ? m[1] : ""));
        const em = m[7] ? Number(m[7]) : (m[8] === "반" ? 30 : Number(m[8] || 0));
        if (!m[5] && m[1] === "오후" && Number(m[6]) < 12 && eh < sh) eh = Math.min(23, Number(m[6]) + 12);
        out.startTime = `${pad2(sh)}:${pad2(sm)}`;
        out.endTime = `${pad2(eh)}:${pad2(em)}`;
        text = text.replace(m[0], " ");
      } else if ((m = text.match(compactTimeRe))) {
        const sh = to24h(m[2], m[1]);
        let eh = to24h(m[4], m[3] || (m[1] && Number(m[4]) < 12 ? m[1] : ""));
        out.startTime = `${pad2(sh)}:00`;
        out.endTime = `${pad2(eh)}:00`;
        text = text.replace(m[0], " ");
      } else {
        // 6b. Separate departure / return times
        const depRe = /(오전|오후|저녁|밤|새벽)?\s*(\d{1,2})\s*(?::(\d{2})|시\s*(\d{1,2}|반)?\s*분?)\s*(?:출타|출발|시작|부터)/;
        const retRe = /(오전|오후|저녁|밤|새벽)?\s*(\d{1,2})\s*(?::(\d{2})|시\s*(\d{1,2}|반)?\s*분?)\s*(?:복귀|귀대|종료|까지)/;
        if ((m = text.match(depRe))) {
          const sh = to24h(m[2], m[1]);
          const sm = m[3] ? Number(m[3]) : (m[4] === "반" ? 30 : Number(m[4] || 0));
          out.startTime = `${pad2(sh)}:${pad2(sm)}`;
          text = text.replace(m[0], " ");
        }
        if ((m = text.match(retRe))) {
          const eh = to24h(m[2], m[1]);
          const em = m[3] ? Number(m[3]) : (m[4] === "반" ? 30 : Number(m[4] || 0));
          out.endTime = `${pad2(eh)}:${pad2(em)}`;
          text = text.replace(m[0], " ");
        }
      }


      // 7. Status (draft / confirmed)
      if (/(가안|미정|조율|\(\?\))/.test(text)) {
        out.status = "draft";
        text = text.replace(/\(?\s*(?:가안|미정|조율|\?)\s*\)?/g, " ");
      } else if (/(?:^|\s)확정(?=\s|$)/.test(text)) {
        out.status = "confirmed";
        text = text.replace(/(?:^|\s)확정(?=\s|$)/g, " ");
      }


      // 8. Kind & Subtype
      const isWork = /(알바|대타|근무|약속|개인|회의|시험|과제|수업)/.test(text);
      out.kind = isWork ? "work" : "outing";
      if (isWork) {
        if (/알바/.test(text)) out.subtype = "알바";
        else if (/대타/.test(text)) out.subtype = "대타";
        else if (/약속/.test(text)) out.subtype = "약속";
        else out.subtype = "기타";
      } else {
        if (/정기/.test(text)) out.subtype = "정기휴가";
        else if (/포상/.test(text)) out.subtype = "포상휴가";
        else if (/(위로|청원|신병|보상)/.test(text)) out.subtype = "위로/청원";
        else if (/외박/.test(text)) out.subtype = "외박";
        else if (/외출/.test(text)) out.subtype = "외출";
        else out.subtype = "기타 출타";
      }


      // 9. Finalize endDate
      if (!out.endDate) {
        if (durationDays) {
          out.endDate = addDays(out.startDate, durationDays - 1);
        } else if (out.kind === "outing" && out.subtype === "외박") {
          out.endDate = addDays(out.startDate, 1);
        } else {
          out.endDate = out.startDate;
        }
      }
      if (out.endDate < out.startDate) out.endDate = out.startDate;


      // 10. Clean remaining title
      const cleaned = text.replace(/\s+/g, " ").replace(/^[~\-,.\s]+|[~\-,.\s]+$/g, "").trim();
      out.title = cleaned || out.subtype;


      return out;
    }

    function safeDocKey(raw) {
      return encodeURIComponent(String(raw || "gcal")).replace(/[^a-zA-Z0-9_-]/g, "_").slice(0, 120);

    async function syncGoogleCalendar(manual = false) {
      state.gcalState.loading = true;
      if (manual) render();
      try {
        const resp = await fetch(`/api/sync${manual ? "?fresh=1" : ""}`);
        if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
        const data = await resp.json();
        state.gcalState.configured = Boolean(data.gcalConfigured);
        state.gcalState.discordConfigured = Boolean(data.discordConfigured);
        state.gcalState.fetchedAt = data.fetchedAt || Date.now();
        state.gcalState.error = data.ok ? "" : (data.error || "동기화 오류");
        if (Array.isArray(data.events)) {
          state.gcalRaw = data.events;
        }
        if (manual) {
          if (!data.gcalConfigured) showToast("Vercel에 GCAL_ICS_URL이 아직 설정되지 않았습니다");
          else if (data.ok) showToast(`구글 캘린더 동기화 완료 (${state.gcalRaw.length}건)`);
          else showToast(`구글 동기화 실패: ${state.gcalState.error}`);
        }
      } catch (err) {
        state.gcalState.error = "API 미배포 (/api/sync)";
        if (manual) showToast("연동 API(/api/sync)를 찾을 수 없습니다");
      } finally {
        state.gcalState.loading = false;
        render();
        handleDeepLinkIfReady();
      }
    }


    async function notifyDiscord({ action, actor, summary, detail = "", targetDate = "", eventId = "" }) {
      const isCore = ["CREATE", "UPDATE", "DELETE", "TEST"].includes(action);
      if (action !== "TEST") {
        if (!state.roomConfig.discordEnabled) return { ok: false, skipped: true };
        if (!isCore && !state.roomConfig.discordNotifySocial) return { ok: false, skipped: true };
      }
      const linkUrl = buildRoomUrl({
        date: targetDate || "",
        ev: action !== "DELETE" && eventId ? eventId : ""
      });
      try {
        const resp = await fetch("/api/sync", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            action,
            actor: actor || state.profile.name || "익명",
            summary,
            detail,
            targetDate,
            roomId: state.roomId,
            url: linkUrl
          })
        });
        return await resp.json();
      } catch (err) {
        return { ok: false, error: err.message || "전송 실패" };
      }
    }


    function connectRoom(roomId) {
      unsubs.forEach((u) => { try { u(); } catch {} });
      unsubs = [];
      state.roomId = roomId;
      state.events = [];
      state.doodles = {};
      state.logs = [];
      state.gcalSocial = {};
      state.roomConfig = { gcalAuthor: "", gcalColor: "#FF6B00", gcalEnabled: roomId === "main", discordEnabled: true, discordNotifySocial: false };
      setSyncStatus("wait");


      if (state.deepLinkHandled || (!state.pendingDeepDate && !state.pendingDeepEv)) {
        window.history.replaceState({}, "", buildRoomUrl());
      }


      const eventsCol = collection(db, "rooms", roomId, "events");
      const doodlesCol = collection(db, "rooms", roomId, "doodles");
      const logsCol = collection(db, "rooms", roomId, "logs");
      const gcalSocialCol = collection(db, "rooms", roomId, "gcalSocial");
      const configDoc = doc(db, "rooms", roomId, "meta", "config");


      const handleErr = (err) => {
        console.error("Firestore error:", err);
        setSyncStatus("err", err.message || err.code || "권한 또는 네트워크 오류");
      };


      unsubs.push(onSnapshot(eventsCol, (snap) => {

        arr.sort((a, b) => (a.startDate || "").localeCompare(b.startDate || "") || (a.startTime || "99:99").localeCompare(b.startTime || "99:99") || (a.createdAtMs || 0) - (b.createdAtMs || 0));
        state.events = arr;
        setSyncStatus("live");
        render();
        handleDeepLinkIfReady();
      }, handleErr));


      unsubs.push(onSnapshot(doodlesCol, (snap) => {
        const map = {};
        snap.forEach((d) => { map[d.id] = d.data(); });
        state.doodles = map;
        render();
      }, handleErr));


      unsubs.push(onSnapshot(query(logsCol, orderBy("createdAtMs", "desc"), limit(80)), (snap) => {
        const arr = [];
        snap.forEach((d) => arr.push({ id: d.id, ...d.data() }));
        state.logs = arr;
        render();
      }, handleErr));


      unsubs.push(onSnapshot(gcalSocialCol, (snap) => {
        const map = {};
        snap.forEach((d) => { map[d.id] = d.data(); });
        state.gcalSocial = map;
        render();
      }, () => {}));

          render();
        }
      }, () => {}));
    }

    function getDayMatchLevel(ymd) {
      if (!state.highlightOverlap) return "";
      const visible = getAllEvents().filter((ev) => !state.hiddenPeople.has(ev.author) && ev.startDate <= ymd && ev.endDate >= ymd);
      const outPeople = new Set();
      const workPeople = new Set();
      for (const ev of visible) {
        if (ev.kind === "outing") outPeople.add(ev.author);
        else workPeople.add(ev.author);
      }
      if (outPeople.size >= 2 && workPeople.size === 0) return "best";
      if (outPeople.size >= 1 && workPeople.size === 0) return "good";
      if (outPeople.size >= 2 && workPeople.size > 0) return "partial";
      return "";
    }

    function getMonthWeeks(year, month) {
      const first = new Date(year, month, 1);
      const last = new Date(year, month + 1, 0);
      const start = new Date(first);
      start.setDate(first.getDate() - first.getDay());
      const end = new Date(last);
      end.setDate(last.getDate() + (6 - last.getDay()));


      const weeks = [];
      const cur = new Date(start);
      while (cur <= end) {
        const week = [];
        for (let i = 0; i < 7; i++) {
          week.push({
            ymd: toYMD(cur),
            day: cur.getDate(),
            dow: cur.getDay(),
            inMonth: cur.getMonth() === month
          });
          cur.setDate(cur.getDate() + 1);
        }
        weeks.push(week);
      }
      return weeks;
    }


    function layoutWeekEvents(week, events) {
      const wStart = week[0].ymd;
      const wEnd = week[6].ymd;
      const relevant = events.filter((ev) => ev.startDate <= wEnd && ev.endDate >= wStart);
      relevant.sort((a, b) => {
        if (a.kind !== b.kind) return a.kind === "outing" ? -1 : 1;
        const da = diffDays(a.startDate, a.endDate);
        const db = diffDays(b.startDate, b.endDate);
        if (db !== da) return db - da;
        return a.startDate.localeCompare(b.startDate) || (a.startTime || "99:99").localeCompare(b.startTime || "99:99");
      });


      const lanes = [];
      const overflowCounts = [0, 0, 0, 0, 0, 0, 0];
      const MAX_LANES = 4;


      for (const ev of relevant) {
        const sCol = ev.startDate <= wStart ? 0 : diffDays(wStart, ev.startDate);
        const eCol = ev.endDate >= wEnd ? 6 : diffDays(wStart, ev.endDate);
        let placed = false;


        for (let l = 0; l < lanes.length; l++) {
          let free = true;
          for (let c = sCol; c <= eCol; c++) {
            if (lanes[l].occ[c]) { free = false; break; }
          }
          if (free) {
            for (let c = sCol; c <= eCol; c++) lanes[l].occ[c] = true;
            lanes[l].items.push({
              ev, sCol, eCol,
              contLeft: ev.startDate < wStart,
              contRight: ev.endDate > wEnd
            });
            placed = true;
            break;
          }
        }
        if (!placed) {
          if (lanes.length < MAX_LANES) {
            const occ = Array(7).fill(false);
            for (let c = sCol; c <= eCol; c++) occ[c] = true;
            lanes.push({
              occ,
              items: [{
                ev, sCol, eCol,
                contLeft: ev.startDate < wStart,
                contRight: ev.endDate > wEnd
              }]
            });
          } else {
            for (let c = sCol; c <= eCol; c++) overflowCounts[c]++;
          }
        }
      }
      return { lanes, overflowCounts };
    }

    function computeActiveOutingStatus(e, today) {
      const nowMs = Date.now();
      const hasStart = Boolean(e.startTime);
      const hasEnd = Boolean(e.endTime);


      if (e.startDate === today && hasStart) {
        const sMs = parseDateTimeMs(e.startDate, e.startTime);
        if (nowMs < sMs) {
          const mins = Math.ceil((sMs - nowMs) / 60000);
          return {
            tag: `오늘 ${e.startTime.slice(0, 5)} 출타`,
            sub: `${fmtRemainMins(mins)} 후 출타`,
            pct: 0
          };
        }
      }


      if (e.endDate === today && hasEnd) {
        const eMs = parseDateTimeMs(e.endDate, e.endTime);
        const sMs = parseDateTimeMs(e.startDate, e.startTime || "08:30");
        const pct = Math.max(0, Math.min(100, Math.round(((nowMs - sMs) / Math.max(60000, eMs - sMs)) * 100)));
        if (nowMs >= eMs) {
          return { tag: "복귀 완료", sub: `${e.endTime.slice(0, 5)} 복귀`, pct: 100 };
        }
        const mins = Math.ceil((eMs - nowMs) / 60000);
        return {
          tag: `복귀 ${fmtRemainMins(mins)} 전`,
          sub: `오늘 ${e.endTime.slice(0, 5)} 복귀 (${pct}%)`,
          pct
        };
      }


      const total = Math.max(1, diffDays(e.startDate, e.endDate) + 1);
      const elapsed = Math.min(total, diffDays(e.startDate, today) + 1);
      let pct = Math.round((elapsed / total) * 100);
      if (hasStart || hasEnd) {
        const sMs = parseDateTimeMs(e.startDate, e.startTime || "08:30");
        const eMs = parseDateTimeMs(e.endDate, e.endTime || "21:00");
        pct = Math.max(1, Math.min(99, Math.round(((nowMs - sMs) / Math.max(60000, eMs - sMs)) * 100)));
      }
      const left = diffDays(today, e.endDate);
      return {
        tag: left === 0 ? "오늘 복귀" : `복귀 D-${left}`,
        sub: `${elapsed}/${total}일차 (${pct}%)${hasEnd ? " · " + e.endTime.slice(0, 5) + " 복귀" : ""}`,
        pct
      };
    }

    function renderMonthGrid(year, month, events, showHeader = false) {

      const headerHtml = showHeader ? `
        <div class="scroll-month-header tabular">
          <span>${year}년 ${month + 1}월</span>

            const mStr = `${year}-${pad2(month + 1)}`;

          }).length}건</span>
        </div>
      ` : "";


      const weeksHtml = weeks.map((week) => {
        const { lanes, overflowCounts } = layoutWeekEvents(week, events);

    function computeRecommendations() {
      const startBase = addDays(todayYMD(), 1);
      const myName = state.profile.name;
      const daysLen = state.recDays;
      const candidates = [];


      for (let offset = 0; offset < state.recHorizon; offset++) {
        const s = addDays(startBase, offset);
        const e = addDays(s, daysLen - 1);
        const outPeople = new Set();
        const workPeople = new Set();
        let targetOutOverlapDays = 0;
        let targetWorkConflictDays = 0;
        let myConflict = false;
        let weekendDays = 0;


        for (let k = 0; k < daysLen; k++) {
          const cur = addDays(s, k);
          const dow = parseYMD(cur).getDay();
          if (dow === 0 || dow === 6) weekendDays++;


          for (const ev of getAllEvents()) {
            if (ev.startDate <= cur && ev.endDate >= cur) {
              if (myName && ev.author === myName) myConflict = true;
              const isTarget = state.recTargets.size === 0 || state.recTargets.has(ev.author);
              if (ev.author !== myName) {
                if (ev.kind === "outing") {
                  outPeople.add(ev.author);
                  if (isTarget) targetOutOverlapDays++;
                } else {
                  workPeople.add(ev.author);
                  if (isTarget) targetWorkConflictDays++;
                }
              }
            }
          }
        }


        if (myConflict) continue;


        let score = 0;
        if (state.recStrategy === "max_meet") {
          score = outPeople.size * 30 + targetOutOverlapDays * 10 - workPeople.size * 12 - targetWorkConflictDays * 4 + weekendDays * 4;
        } else if (state.recStrategy === "target_people") {
          score = targetOutOverlapDays * 25 - targetWorkConflictDays * 18 + weekendDays * 3;
        } else {
          score = 100 - (outPeople.size * 20 + workPeople.size * 15) - weekendDays * 2;
        }


        candidates.push({
          startDate: s,
          endDate: e,
          score,
          outPeople: Array.from(outPeople),
          workPeople: Array.from(workPeople),
          weekendDays
        });
      }


      candidates.sort((a, b) => b.score - a.score || a.startDate.localeCompare(b.startDate));
      const picked = [];
      for (const c of candidates) {
        if (picked.length >= 5) break;
        const tooClose = picked.some((p) => Math.abs(diffDays(p.startDate, c.startDate)) < Math.max(2, Math.floor(daysLen / 2)));
        if (!tooClose) picked.push(c);
      }
      return picked;
    }

    function attachSwipeHandlers() {
      const zone = document.getElementById("cal-swipe-zone");
      if (!zone) return;
      let sx = 0, sy = 0, st = 0;
      zone.addEventListener("touchstart", (e) => {
        if (e.touches.length !== 1) return;
        sx = e.touches[0].clientX;
        sy = e.touches[0].clientY;
        st = Date.now();
      }, { passive: true });
      zone.addEventListener("touchend", (e) => {
        if (e.changedTouches.length !== 1) return;
        const dx = e.changedTouches[0].clientX - sx;
        const dy = e.changedTouches[0].clientY - sy;
        const dt = Date.now() - st;
        if (dt < 420 && Math.abs(dx) > 58 && Math.abs(dx) > Math.abs(dy) * 1.4) {
          shiftMonth(dx < 0 ? 1 : -1);
        }
      }, { passive: true });
    }


    let longPressTimer = null;

    function attachDayLongPress() {
      const cells = document.querySelectorAll(".day-bg-cell[data-day]");
      cells.forEach((cell) => {
        const ymd = cell.dataset.day;
        const start = () => {
          longPressFired = false;
          clearTimeout(longPressTimer);
          longPressTimer = setTimeout(() => {
            longPressFired = true;
            if (navigator.vibrate) try { navigator.vibrate(18); } catch {}
            openEventFormSheet({ startDate: ymd, endDate: ymd });
          }, 460);
        };
        const cancel = () => clearTimeout(longPressTimer);
        cell.addEventListener("touchstart", start, { passive: true });
        cell.addEventListener("touchmove", cancel, { passive: true });
        cell.addEventListener("touchend", cancel, { passive: true });
        cell.addEventListener("mousedown", start);
        cell.addEventListener("mouseleave", cancel);
        cell.addEventListener("mouseup", cancel);
      });
    }

        const pt = e.touches ? e.touches[0] : e;
        return { x: pt.clientX - r.left, y: pt.clientY - r.top };
      };
      const startDraw = (e) => {
        e.preventDefault();
        drawing = true;
        hasStroke = true;
        const p = getPos(e);
        ctx.strokeStyle = penColor;
        ctx.lineWidth = sizes[sizeIdx].px;
        ctx.beginPath();
        ctx.moveTo(p.x, p.y);
      };
      const moveDraw = (e) => {
        if (!drawing) return;
        e.preventDefault();
        const p = getPos(e);
        ctx.lineTo(p.x, p.y);
        ctx.stroke();
      };
      const endDraw = () => { drawing = false; };


      canvas.addEventListener("mousedown", startDraw);
      canvas.addEventListener("mousemove", moveDraw);
      window.addEventListener("mouseup", endDraw);
      canvas.addEventListener("touchstart", startDraw, { passive: false });
      canvas.addEventListener("touchmove", moveDraw, { passive: false });
      canvas.addEventListener("touchend", endDraw);


      document.querySelectorAll("[data-pen-color]").forEach((btn) => {
        btn.onclick = () => {
          penColor = btn.dataset.penColor;
          document.querySelectorAll("[data-pen-color]").forEach((b) => b.classList.remove("active"));
          btn.classList.add("active");
        };
      });
      document.getElementById("btn-pen-size").onclick = (e) => {
        sizeIdx = (sizeIdx + 1) % sizes.length;
        e.target.textContent = `굵기: ${sizes[sizeIdx].label}`;
      };
      document.getElementById("btn-canvas-clear").onclick = () => {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        hasStroke = false;
      };
      document.getElementById("btn-canvas-save").onclick = () => {
        const memo = document.getElementById("day-memo-input").value.trim();
        const imgData = hasStroke ? canvas.toDataURL("image/png") : "";
        saveDayData(memo, imgData);
      };
    }

                targetDate: payload.startDate,
                eventId: formState.id
              });
              showToast("일정이 수정되었습니다");

              const weeks = formState.kind === "work" ? formState.repeatWeeks : 1;

                  ...payload,
                  startDate: sD,
                  endDate: eD,
                  rsvps: { ok: [], maybe: [], no: [] },
                  comments: [],
                  createdAtMs: Date.now() + w
                });

                action: "CREATE",
                actor: state.profile.name || author,
                summary: `${payload.author} · ${payload.title} (${rangeLabel})${weeks > 1 ? ` ${weeks}주 반복` : ""} 등록`,
                detail: payload.status === "draft" ? "가안으로 등록됨" : "",
                targetDate: payload.startDate,
                eventId: firstDocId
              });

            }
            closeSheet();
          } catch (err) {
            showToast("저장 오류: " + (err.message || "권한 확인 필요"));
          }
        };
      }
      renderForm();
    }

    function shiftMonth(delta) {
      const d = new Date(state.calYear, state.calMonth + delta, 1);
      state.calYear = d.getFullYear();
      state.calMonth = d.getMonth();
      render();
    }


    async function copyText(str, msg) {
      try {
        await navigator.clipboard.writeText(str);
        showToast(msg);
      } catch {
        const ta = document.createElement("textarea");
        ta.value = str;
        document.body.appendChild(ta);
        ta.select();
        document.execCommand("copy");
        document.body.removeChild(ta);
        showToast(msg);
      }
    }


    function buildMonthSummaryText() {
      const mStr = `${state.calYear}-${pad2(state.calMonth + 1)}`;
      const evs = getFilteredEvents().filter((e) => e.startDate.slice(0, 7) <= mStr && e.endDate.slice(0, 7) >= mStr);
      const lines = [`[chulta4] ${state.calYear}년 ${state.calMonth + 1}월 공유 일정`];
      if (evs.length === 0) {
        lines.push("• 등록된 일정 없음");
      } else {
        evs.forEach((e) => {
          const tag = e.status === "draft" ? "(가안) " : "";
          lines.push(`• ${fmtEventFullRange(e)} | ${e.author} — ${tag}${e.title || e.subtype} (${fmtDuration(e.startDate, e.endDate)})`);
        });
      }
      lines.push("", `실시간 달력: ${buildRoomUrl()}`);
      return lines.join("\n");
    }
