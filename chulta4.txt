    const pad2 = (n) => String(n).padStart(2, "0");
    function toYMD(d) {
      return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}`;
    }
    function parseYMD(s) {
      const [y, m, d] = s.split("-").map(Number);
      return new Date(y, m - 1, d);
    }
    function addDays(ymd, delta) {
      const d = parseYMD(ymd);
      d.setDate(d.getDate() + delta);
      return toYMD(d);
    }
    function diffDays(a, b) {
      const ms = parseYMD(b).getTime() - parseYMD(a).getTime();
      return Math.round(ms / 86400000);
    }
    function todayYMD() {
      return toYMD(new Date());
    }
    const WEEKDAYS = ["일", "월", "화", "수", "목", "금", "토"];
    function fmtShort(ymd) {
      const d = parseYMD(ymd);
      return `${d.getMonth() + 1}/${d.getDate()}(${WEEKDAYS[d.getDay()]})`;
    }
    function fmtRange(s, e) {
      if (s === e) return fmtShort(s);
      return `${fmtShort(s)} ~ ${fmtShort(e)}`;
    }
    function fmtDuration(s, e) {
      const days = diffDays(s, e) + 1;
      if (days <= 1) return "당일";
      return `${days - 1}박 ${days}일`;
    }
    function fmtRelative(isoOrMs) {
      if (!isoOrMs) return "방금";
      const t = typeof isoOrMs === "number" ? isoOrMs : new Date(isoOrMs).getTime();
      const diff = Math.max(0, Math.floor((Date.now() - t) / 1000));
      if (diff < 60) return "방금 전";
      if (diff < 3600) return `${Math.floor(diff / 60)}분 전`;
      if (diff < 86400) return `${Math.floor(diff / 3600)}시간 전`;
      if (diff < 86400 * 7) return `${Math.floor(diff / 86400)}일 전`;
      const d = new Date(t);
      return `${d.getMonth() + 1}/${d.getDate()}`;
    }

    function connectRoom(roomId) {
      unsubs.forEach((u) => { try { u(); } catch {} });
      unsubs = [];
      state.roomId = roomId;
      state.events = [];
      state.doodles = {};
      state.logs = [];
      setSyncStatus("wait");


      const url = new URL(window.location.href);
      if (roomId === "main") url.searchParams.delete("room");
      else url.searchParams.set("room", roomId);
      window.history.replaceState({}, "", url.toString());


      const eventsCol = collection(db, "rooms", roomId, "events");
      const doodlesCol = collection(db, "rooms", roomId, "doodles");
      const logsCol = collection(db, "rooms", roomId, "logs");


      const handleErr = (err) => {
        console.error("Firestore error:", err);
        setSyncStatus("err", err.message || err.code || "권한 또는 네트워크 오류");
      };


      unsubs.push(onSnapshot(eventsCol, (snap) => {
        const arr = [];
        snap.forEach((d) => arr.push({ id: d.id, ...d.data() }));
        arr.sort((a, b) => (a.startDate || "").localeCompare(b.startDate || "") || (a.createdAtMs || 0) - (b.createdAtMs || 0));
        state.events = arr;
        setSyncStatus("live");
        render();
      }, handleErr));


      unsubs.push(onSnapshot(doodlesCol, (snap) => {
        const map = {};
        snap.forEach((d) => { map[d.id] = d.data(); });
        state.doodles = map;
        render();
      }, handleErr));


      unsubs.push(onSnapshot(query(logsCol, orderBy("createdAtMs", "desc"), limit(80)), (snap) => {
        const arr = [];
        snap.forEach((d) => arr.push({ id: d.id, ...d.data() }));
        state.logs = arr;
        render();
      }, handleErr));
    }

    // =========================================================================
    // Spanning Calendar Layout Algorithm
    // =========================================================================
    function getMonthWeeks(year, month) {
      const first = new Date(year, month, 1);
      const last = new Date(year, month + 1, 0);
      const start = new Date(first);
      start.setDate(first.getDate() - first.getDay());
      const end = new Date(last);
      end.setDate(last.getDate() + (6 - last.getDay()));


      const weeks = [];
      const cur = new Date(start);
      while (cur <= end) {
        const week = [];
        for (let i = 0; i < 7; i++) {
          week.push({
            ymd: toYMD(cur),
            day: cur.getDate(),
            dow: cur.getDay(),
            inMonth: cur.getMonth() === month
          });
          cur.setDate(cur.getDate() + 1);
        }
        weeks.push(week);
      }
      return weeks;
    }


    function layoutWeekEvents(week, events) {
      const wStart = week[0].ymd;
      const wEnd = week[6].ymd;
      const relevant = events.filter((ev) => ev.startDate <= wEnd && ev.endDate >= wStart);
      relevant.sort((a, b) => {
        if (a.kind !== b.kind) return a.kind === "outing" ? -1 : 1;
        const da = diffDays(a.startDate, a.endDate);
        const db = diffDays(b.startDate, b.endDate);
        if (db !== da) return db - da;
        return a.startDate.localeCompare(b.startDate);
      });


      const lanes = [];
      const overflowCounts = [0, 0, 0, 0, 0, 0, 0];
      const MAX_LANES = 4;


      for (const ev of relevant) {
        const sCol = ev.startDate <= wStart ? 0 : diffDays(wStart, ev.startDate);
        const eCol = ev.endDate >= wEnd ? 6 : diffDays(wStart, ev.endDate);
        let placed = false;


        for (let l = 0; l < lanes.length; l++) {
          let free = true;
          for (let c = sCol; c <= eCol; c++) {
            if (lanes[l].occ[c]) { free = false; break; }
          }
          if (free) {
            for (let c = sCol; c <= eCol; c++) lanes[l].occ[c] = true;
            lanes[l].items.push({
              ev, sCol, eCol,
              contLeft: ev.startDate < wStart,
              contRight: ev.endDate > wEnd
            });
            placed = true;
            break;
          }
        }
        if (!placed) {
          if (lanes.length < MAX_LANES) {
            const occ = Array(7).fill(false);
            for (let c = sCol; c <= eCol; c++) occ[c] = true;
            lanes.push({
              occ,
              items: [{
                ev, sCol, eCol,
                contLeft: ev.startDate < wStart,
                contRight: ev.endDate > wEnd
              }]
            });
          } else {
            for (let c = sCol; c <= eCol; c++) overflowCounts[c]++;
          }
        }
      }
      return { lanes, overflowCounts };
    }

    function renderMiniTicker() {
      const today = todayYMD();
      const outings = state.events.filter((e) => e.kind === "outing");
      const activeNow = outings.filter((e) => e.startDate <= today && e.endDate >= today);
      const mine = state.profile.name;


      // Prioritize my active -> any active -> my next -> any next
      const myActive = mine ? activeNow.find((e) => e.author === mine) : null;
      const upcoming = outings.filter((e) => e.startDate > today).sort((a, b) => a.startDate.localeCompare(b.startDate));
      const myNext = mine ? upcoming.find((e) => e.author === mine) : null;


      let tag = "D-DAY";
      let text = "등록된 출타 일정이 없습니다 · + 버튼으로 추가해 보세요";
      if (myActive) {
        const left = diffDays(today, myActive.endDate);
        tag = left === 0 ? "오늘 복귀" : `복귀 D-${left}`;
        text = `${myActive.author} · ${myActive.title || myActive.subtype} (${fmtRange(myActive.startDate, myActive.endDate)})`;
      } else if (myNext) {
        const left = diffDays(today, myNext.startDate);
        tag = `출타 D-${left}`;
        text = `${myNext.author} · ${myNext.title || myNext.subtype} (${fmtShort(myNext.startDate)} · ${fmtDuration(myNext.startDate, myNext.endDate)})`;
      } else if (activeNow.length > 0) {
        const first = activeNow[0];
        const left = diffDays(today, first.endDate);
        tag = left === 0 ? "오늘 복귀" : `복귀 D-${left}`;
        text = `${first.author} · ${first.title || first.subtype}`;
      } else if (upcoming.length > 0) {
        const first = upcoming[0];
        const left = diffDays(today, first.startDate);
        tag = `출타 D-${left}`;
        text = `${first.author} · ${first.title || first.subtype} (${fmtShort(first.startDate)})`;
      }


      const rightText = activeNow.length > 0 ? `오늘 출타 ${activeNow.length}명 ›` : `디데이 ›`;
      return `
        <div class="dday-ticker" id="mini-ticker">
          <div class="ticker-left">
            <span class="ticker-tag tabular">${esc(tag)}</span>
            <span class="ticker-text">${esc(text)}</span>
          </div>
          <span class="ticker-right tabular">${esc(rightText)}</span>
        </div>
      `;
    }

    function renderMonthGrid(year, month, events, showHeader = false) {
      const weeks = getMonthWeeks(year, month);
      const today = todayYMD();


      const headerHtml = showHeader ? `
        <div class="scroll-month-header tabular">
          <span>${year}년 ${month + 1}월</span>
          <span style="font-size:11.5px;color:var(--text-3);font-weight:600;">${events.filter((e) => {
            const mStr = `${year}-${pad2(month + 1)}`;
            return e.startDate.slice(0, 7) <= mStr && e.endDate.slice(0, 7) >= mStr;
          }).length}건</span>
        </div>
      ` : "";


      const weeksHtml = weeks.map((week) => {
        const { lanes, overflowCounts } = layoutWeekEvents(week, events);
        const bgCells = week.map((d) => {
          const doodle = state.doodles[d.ymd];
          const hasImg = state.showDoodles && doodle && doodle.image;
          return `
            <div class="day-bg-cell ${d.inMonth ? "" : "other-month"} ${d.ymd === state.selectedDay ? "selected-day" : ""}" data-day="${d.ymd}">
              ${hasImg ? `<img class="doodle-thumb" src="${doodle.image}" alt="" />` : ""}
            </div>
          `;
        }).join("");


        const numCells = week.map((d) => {
          const doodle = state.doodles[d.ymd];
          const hasNote = doodle && (doodle.image || (doodle.memo && doodle.memo.trim()));
          const dowCls = d.dow === 0 ? "sun" : (d.dow === 6 ? "sat" : "");
          return `
            <div class="day-num-wrap">
              <span class="day-num tabular ${dowCls} ${d.inMonth ? "" : "other"} ${d.ymd === today ? "today" : ""}">${d.day}</span>
              ${hasNote ? `<span class="day-note-dot" title="메모/낙서 있음"></span>` : ""}
            </div>
          `;
        }).join("");


        const lanesHtml = lanes.map((lane) => {
          const bars = lane.items.map(({ ev, sCol, eCol, contLeft, contRight }) => {
            const color = ev.color || "#FF6B00";
            const kindCls = ev.kind === "work" ? "kind-work" : "kind-outing";
            const draftCls = ev.status === "draft" ? "is-draft" : "";
            const labelText = ev.kind === "work"
              ? `${ev.author}${ev.startTime ? " " + ev.startTime.slice(0, 5) : ""} ${ev.title || ev.subtype || ""}`.trim()
              : `${ev.author} ${ev.title || ev.subtype || ""}`.trim();
            return `
              <div class="ev-bar ${kindCls} ${draftCls} ${contLeft ? "cont-left" : ""} ${contRight ? "cont-right" : ""}"
                   style="grid-column: ${sCol + 1} / ${eCol + 2}; background:${esc(color)}; --ev-color:${esc(color)};"
                   data-ev-id="${esc(ev.id)}"
                   title="${esc(labelText)}">
                ${ev.status === "draft" ? `<span style="opacity:.85">[가안]</span>` : ""}
                <span>${esc(labelText)}</span>
              </div>
            `;
          }).join("");
          return `<div class="bar-lane">${bars}</div>`;
        }).join("");


        const hasOverflow = overflowCounts.some((n) => n > 0);
        const overflowHtml = hasOverflow ? `
          <div class="more-row">
            ${overflowCounts.map((n) => `<div class="more-cell tabular">${n > 0 ? `+${n}` : ""}</div>`).join("")}
          </div>
        ` : "";


        return `
          <div class="week-row">
            <div class="week-bg-grid">${bgCells}</div>
            <div class="week-header-grid">${numCells}</div>
            <div class="week-bars">${lanesHtml}</div>
            ${overflowHtml}
          </div>
        `;
      }).join("");


      return `
        <div class="cal-month-block">
          ${headerHtml}
          <div class="weekday-row">
            ${WEEKDAYS.map((w, i) => `<div class="weekday-cell ${i === 0 ? "sun" : (i === 6 ? "sat" : "")}">${w}</div>`).join("")}
          </div>
          ${weeksHtml}
        </div>
      `;
    }

      const people = Array.from(peopleMap.entries()).map(([name, color]) => ({ name, color }));
      const myName = state.profile.name;


      const filtered = allOutings.filter((e) => !state.ddayHiddenPeople.has(e.author));

      const past = filtered.filter((e) => e.endDate < today).sort((a, b) => b.endDate.localeCompare(a.endDate));

      const lastReturnByAuthor = new Map();

    function computeRecommendations() {
      const startBase = addDays(todayYMD(), 1);
      const myName = state.profile.name;
      const daysLen = state.recDays;
      const candidates = [];


      for (let offset = 0; offset < state.recHorizon; offset++) {
        const s = addDays(startBase, offset);
        const e = addDays(s, daysLen - 1);
        const outPeople = new Set();
        const workPeople = new Set();
        let targetOutOverlapDays = 0;
        let targetWorkConflictDays = 0;
        let myConflict = false;
        let weekendDays = 0;


        for (let k = 0; k < daysLen; k++) {
          const cur = addDays(s, k);
          const dow = parseYMD(cur).getDay();
          if (dow === 0 || dow === 6) weekendDays++;


          for (const ev of state.events) {
            if (ev.startDate <= cur && ev.endDate >= cur) {
              if (myName && ev.author === myName) myConflict = true;
              const isTarget = state.recTargets.size === 0 || state.recTargets.has(ev.author);
              if (ev.author !== myName) {
                if (ev.kind === "outing") {
                  outPeople.add(ev.author);
                  if (isTarget) targetOutOverlapDays++;
                } else {
                  workPeople.add(ev.author);
                  if (isTarget) targetWorkConflictDays++;
                }
              }
            }
          }
        }


        if (myConflict) continue;


        let score = 0;
        if (state.recStrategy === "max_meet") {
          score = outPeople.size * 30 + targetOutOverlapDays * 10 - workPeople.size * 12 - targetWorkConflictDays * 4 + weekendDays * 4;
        } else if (state.recStrategy === "target_people") {
          score = targetOutOverlapDays * 25 - targetWorkConflictDays * 18 + weekendDays * 3;
        } else {
          // quiet
          score = 100 - (outPeople.size * 20 + workPeople.size * 15) - weekendDays * 2;
        }


        candidates.push({
          startDate: s,
          endDate: e,
          score,
          outPeople: Array.from(outPeople),
          workPeople: Array.from(workPeople),
          weekendDays
        });
      }


      candidates.sort((a, b) => b.score - a.score || a.startDate.localeCompare(b.startDate));
      // Pick top 5 non-overlapping (or spaced by at least 2 days)
      const picked = [];
      for (const c of candidates) {
        if (picked.length >= 5) break;
        const tooClose = picked.some((p) => Math.abs(diffDays(p.startDate, c.startDate)) < Math.max(2, Math.floor(daysLen / 2)));
        if (!tooClose) picked.push(c);
      }
      return picked;
    }

    function attachSwipeHandlers() {
      const zone = document.getElementById("cal-swipe-zone");
      if (!zone) return;
      let sx = 0, sy = 0, st = 0;
      zone.addEventListener("touchstart", (e) => {
        if (e.touches.length !== 1) return;
        sx = e.touches[0].clientX;
        sy = e.touches[0].clientY;
        st = Date.now();
      }, { passive: true });
      zone.addEventListener("touchend", (e) => {
        if (e.changedTouches.length !== 1) return;
        const dx = e.changedTouches[0].clientX - sx;
        const dy = e.changedTouches[0].clientY - sy;
        const dt = Date.now() - st;
        if (dt < 420 && Math.abs(dx) > 58 && Math.abs(dx) > Math.abs(dy) * 1.4) {
          shiftMonth(dx < 0 ? 1 : -1);
        }
      }, { passive: true });
    }


    let longPressTimer = null;

    function attachDayLongPress() {
      const cells = document.querySelectorAll(".day-bg-cell[data-day]");
      cells.forEach((cell) => {
        const ymd = cell.dataset.day;
        const start = () => {
          longPressFired = false;
          clearTimeout(longPressTimer);
          longPressTimer = setTimeout(() => {
            longPressFired = true;
            if (navigator.vibrate) try { navigator.vibrate(18); } catch {}
            openEventFormSheet({ startDate: ymd, endDate: ymd });
          }, 460);
        };
        const cancel = () => clearTimeout(longPressTimer);
        cell.addEventListener("touchstart", start, { passive: true });
        cell.addEventListener("touchmove", cancel, { passive: true });
        cell.addEventListener("touchend", cancel, { passive: true });
        cell.addEventListener("mousedown", start);
        cell.addEventListener("mouseleave", cancel);
        cell.addEventListener("mouseup", cancel);
      });
    }

      const sizes = [{ label: "얇게", px: 2.2 }, { label: "중", px: 4 }, { label: "굵게", px: 7.5 }];
      let sizeIdx = 1;
      let drawing = false;


      const getPos = (e) => {
        const r = canvas.getBoundingClientRect();
        const pt = e.touches ? e.touches[0] : e;
        return { x: pt.clientX - r.left, y: pt.clientY - r.top };
      };
      const startDraw = (e) => {
        e.preventDefault();
        drawing = true;
        hasStroke = true;
        const p = getPos(e);
        ctx.strokeStyle = penColor;
        ctx.lineWidth = sizes[sizeIdx].px;
        ctx.beginPath();
        ctx.moveTo(p.x, p.y);
      };
      const moveDraw = (e) => {
        if (!drawing) return;
        e.preventDefault();
        const p = getPos(e);
        ctx.lineTo(p.x, p.y);
        ctx.stroke();
      };
      const endDraw = () => { drawing = false; };


      canvas.addEventListener("mousedown", startDraw);
      canvas.addEventListener("mousemove", moveDraw);
      window.addEventListener("mouseup", endDraw, { once: false });
      canvas.addEventListener("touchstart", startDraw, { passive: false });
      canvas.addEventListener("touchmove", moveDraw, { passive: false });
      canvas.addEventListener("touchend", endDraw);


      document.querySelectorAll("[data-pen-color]").forEach((btn) => {
        btn.onclick = () => {
          penColor = btn.dataset.penColor;
          document.querySelectorAll("[data-pen-color]").forEach((b) => b.classList.remove("active"));
          btn.classList.add("active");
        };
      });
      document.getElementById("btn-pen-size").onclick = (e) => {
        sizeIdx = (sizeIdx + 1) % sizes.length;
        e.target.textContent = `굵기: ${sizes[sizeIdx].label}`;
      };
      document.getElementById("btn-canvas-clear").onclick = () => {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        hasStroke = false;
      };
      document.getElementById("btn-canvas-save").onclick = async () => {
        const memo = document.getElementById("day-memo-input").value.trim();
        const imgData = hasStroke ? canvas.toDataURL("image/png") : "";
        const ref = doc(db, "rooms", state.roomId, "doodles", ymd);
        try {
          if (!memo && !imgData) {
            await deleteDoc(ref);
            showToast("날짜 메모·낙서를 비웠습니다");
          } else {
            await setDoc(ref, {
              memo,
              image: imgData,
              updatedBy: state.profile.name || "익명",
              updatedAtMs: Date.now()
            });
            await writeLog({
              action: "DOODLE",
              actor: state.profile.name || "익명",
              summary: `${fmtShort(ymd)}에 ${imgData && memo ? "메모와 낙서" : (imgData ? "낙서" : `메모("${memo.slice(0, 24)}")`)}를 남겼습니다.`,
              targetDate: ymd
            });
            showToast("메모·낙서가 실시간 저장되었습니다");
          }
          closeSheet();
        } catch (err) {
          showToast("저장 실패: " + (err.message || "오류"));
        }
      };
    }

            renderForm();
          };
        });
        document.querySelectorAll("[data-f-repeat]").forEach((b) => {
          b.onclick = () => { syncInputs(); formState.repeatWeeks = Number(b.dataset.fRepeat); renderForm(); };

          b.onclick = () => { syncInputs(); formState.color = b.dataset.fColor; renderForm(); };

          formState.endDate = addDays(formState.startDate, prevLen);
          renderForm();
        };
        document.getElementById("f-end").onchange = (e) => {
          syncInputs();
          formState.endDate = e.target.value;
          if (formState.endDate < formState.startDate) formState.endDate = formState.startDate;
          renderForm();
        };


        document.getElementById("f-submit").onclick = async () => {
          syncInputs();
          const author = formState.author.trim();
          if (!author) { showToast("이름을 입력해 주세요"); document.getElementById("f-author").focus(); return; }
          if (!formState.startDate || !formState.endDate) { showToast("날짜를 확인해 주세요"); return; }


          rememberProfile(author, formState.color);
          const payload = {
            kind: formState.kind,
            status: formState.status,
            author,
            subtype: formState.subtype,
            title: formState.title.trim() || formState.subtype,
            startDate: formState.startDate,
            endDate: formState.endDate,

            endTime: formState.kind === "work" ? formState.endTime : "",
            color: formState.color,
            note: formState.note.trim()
          };

                  diffs.push(`날짜 ${fmtRange(prev.startDate, prev.endDate)} → ${fmtRange(payload.startDate, payload.endDate)}`);
                }
                if (prev.title !== payload.title) diffs.push(`명칭 "${prev.title}" → "${payload.title}"`);
                if (prev.status !== payload.status) diffs.push(`상태 ${prev.status === "draft" ? "가안" : "확정"} → ${payload.status === "draft" ? "가안" : "확정"}`);
              }
              await writeLog({
                action: "UPDATE",
                actor: state.profile.name || author,
                summary: `${payload.author} · ${payload.title} 일정을 수정했습니다.`,
                detail: diffs.join(" · "),
                targetDate: payload.startDate
              });
              showToast("일정이 수정되었습니다");
            } else {
              const weeks = formState.kind === "work" ? formState.repeatWeeks : 1;
              for (let w = 0; w < weeks; w++) {
                const sD = addDays(payload.startDate, w * 7);
                const eD = addDays(payload.endDate, w * 7);
                await addDoc(collection(db, "rooms", state.roomId, "events"), {
                  ...payload,
                  startDate: sD,
                  endDate: eD,
                  rsvps: { ok: [], maybe: [], no: [] },
                  comments: [],
                  createdAtMs: Date.now() + w
                });
              }
              await writeLog({
                action: "CREATE",
                actor: state.profile.name || author,
                summary: `${payload.author} · ${payload.title} (${fmtRange(payload.startDate, payload.endDate)})${weeks > 1 ? ` ${weeks}주 반복` : ""} 등록`,
                detail: payload.status === "draft" ? "가안으로 등록됨" : "",
                targetDate: payload.startDate
              });
              showToast(weeks > 1 ? `${weeks}주 반복 일정이 등록되었습니다` : "일정이 등록되었습니다");
            }
            closeSheet();
          } catch (err) {
            showToast("저장 오류: " + (err.message || "권한 확인 필요"));
          }
        };
      }
      renderForm();
    }

          const next = {
            ok: [...(rsvps.ok || [])].filter((x) => x !== actor),
            maybe: [...(rsvps.maybe || [])].filter((x) => x !== actor),
            no: [...(rsvps.no || [])].filter((x) => x !== actor)
          };
          const wasInKey = (rsvps[key] || []).includes(actor);
          if (!wasInKey) next[key].push(actor);


          await updateDoc(doc(db, "rooms", state.roomId, "events", ev.id), { rsvps: next });
          if (!wasInKey) {
            const labelMap = { ok: "만남 가능", maybe: "조율 필요", no: "불가" };
            await writeLog({
              action: "RSVP",
              actor,
              summary: `${ev.author} · ${ev.title} 일정에 [${labelMap[key]}] 표시`,
              targetDate: ev.startDate
            });
          }
          setTimeout(() => openEventDetailSheet(ev.id), 180);
        };
      });


      const sendComment = async () => {
        const input = document.getElementById("det-comment-input");
        const text = input.value.trim();
        if (!text) return;
        const actor = ensureName();
        if (!actor) return;
        const nextComments = [...comments, { id: String(Date.now()), author: actor, text, createdAtMs: Date.now() }];
        await updateDoc(doc(db, "rooms", state.roomId, "events", ev.id), { comments: nextComments });
        await writeLog({
          action: "COMMENT",
          actor,
          summary: `${ev.author} · ${ev.title}에 댓글: "${text.slice(0, 40)}"`,
          targetDate: ev.startDate
        });
        input.value = "";
        setTimeout(() => openEventDetailSheet(ev.id), 180);
      };
      document.getElementById("det-comment-send").onclick = sendComment;
      document.getElementById("det-comment-input").onkeydown = (e) => { if (e.key === "Enter") sendComment(); };
    }


    // =========================================================================
    // Actions & Utilities
    // =========================================================================
    function shiftMonth(delta) {
      const d = new Date(state.calYear, state.calMonth + delta, 1);
      state.calYear = d.getFullYear();
      state.calMonth = d.getMonth();
      render();
    }


    async function copyText(str, msg) {
      try {
        await navigator.clipboard.writeText(str);
        showToast(msg);
      } catch {
        const ta = document.createElement("textarea");
        ta.value = str;
        document.body.appendChild(ta);
        ta.select();
        document.execCommand("copy");
        document.body.removeChild(ta);
        showToast(msg);
      }
    }


    function buildMonthSummaryText() {
      const mStr = `${state.calYear}-${pad2(state.calMonth + 1)}`;
      const evs = getFilteredEvents().filter((e) => e.startDate.slice(0, 7) <= mStr && e.endDate.slice(0, 7) >= mStr);
      const lines = [`[chulta4] ${state.calYear}년 ${state.calMonth + 1}월 공유 일정`];
      if (evs.length === 0) {
        lines.push("• 등록된 일정 없음");
      } else {
        evs.forEach((e) => {
          const tag = e.status === "draft" ? "(가안) " : "";
          lines.push(`• ${fmtRange(e.startDate, e.endDate)} | ${e.author} — ${tag}${e.title || e.subtype} (${fmtDuration(e.startDate, e.endDate)})`);
        });
      }
      lines.push("", `실시간 달력: ${window.location.href}`);
      return lines.join("\n");
    }


    // Global Event Delegation
    document.addEventListener("click", (e) => {
      const tabBtn = e.target.closest(".nav-tab[data-tab]");
      if (tabBtn) {
        state.tab = tabBtn.dataset.tab;
        render();
        window.scrollTo({ top: 0, behavior: "instant" });
        return;
      }


      if (e.target.closest("#mini-ticker")) {
        state.tab = "dday";
        render();
        return;
      }


      const evEl = e.target.closest("[data-ev-id]");
      if (evEl) {
        e.stopPropagation();
        openEventDetailSheet(evEl.dataset.evId);
        return;
      }


      const dayCell = e.target.closest(".day-bg-cell[data-day]");
      if (dayCell) {
        if (longPressFired) { longPressFired = false; return; }
        openDaySheet(dayCell.dataset.day);
        return;
      }


      const actEl = e.target.closest("[data-action]");
      if (!actEl) return;
      const act = actEl.dataset.action;


      if (act === "prev-month") shiftMonth(-1);
      else if (act === "next-month") shiftMonth(1);
      else if (act === "go-today") {
        const t = new Date();
        state.calYear = t.getFullYear();
        state.calMonth = t.getMonth();
        state.scrollOffsetStart = 0;
        state.selectedDay = todayYMD();
        render();
      } else if (act === "set-view") {
        state.viewMode = actEl.dataset.val;
        saveLS("chulta4_view_mode", state.viewMode);
        render();
      } else if (act === "scroll-prev-more") {
        state.scrollOffsetStart -= 1;
        state.scrollMonthsCount += 1;
        render();
      } else if (act === "scroll-next-more") {
        state.scrollMonthsCount += 2;
        render();
      } else if (act === "kind-filter") {
        state.kindFilter = actEl.dataset.val;
        render();
      } else if (act === "toggle-doodles") {
        state.showDoodles = !state.showDoodles;
        render();
      } else if (act === "toggle-person") {
        const p = actEl.dataset.person;
        if (state.hiddenPeople.has(p)) state.hiddenPeople.delete(p);
        else state.hiddenPeople.add(p);
        render();
      } else if (act === "people-all") {
        state.hiddenPeople.clear();
        render();
      } else if (act === "people-none") {
        getPeopleList().forEach((p) => state.hiddenPeople.add(p.name));
        render();
      } else if (act === "people-mine") {
        state.hiddenPeople.clear();
        getPeopleList().forEach((p) => { if (p.name !== state.profile.name) state.hiddenPeople.add(p.name); });
        render();
      } else if (act === "dday-all") {
        state.ddayHiddenPeople.clear();
        render();
      } else if (act === "dday-mine") {
        state.ddayHiddenPeople.clear();
        getPeopleList().forEach((p) => { if (p.name !== state.profile.name) state.ddayHiddenPeople.add(p.name); });
        render();
      } else if (act === "dday-toggle-person") {
        const p = actEl.dataset.person;
        if (state.ddayHiddenPeople.has(p)) state.ddayHiddenPeople.delete(p);
        else state.ddayHiddenPeople.add(p);
        render();
      } else if (act === "toggle-past") {
        state.showPastOutings = !state.showPastOutings;
        render();
      } else if (act === "open-add-outing") {
        openEventFormSheet({ kind: "outing" });
      } else if (act === "rec-days") {
        state.recDays = Number(actEl.dataset.val);
        render();
      } else if (act === "rec-strategy") {
        state.recStrategy = actEl.dataset.val;
        render();
      } else if (act === "rec-horizon") {
        state.recHorizon = Number(actEl.dataset.val);
        render();
      } else if (act === "rec-toggle-target") {
        const p = actEl.dataset.person;
        if (state.recTargets.has(p)) state.recTargets.delete(p);
        else state.recTargets.add(p);
        render();
      } else if (act === "rec-apply") {
        openEventFormSheet({
          kind: "outing",
          startDate: actEl.dataset.start,
          endDate: actEl.dataset.end,
          status: actEl.dataset.status
        });
      } else if (act === "activity-sub") {
        state.activitySub = actEl.dataset.val;
        render();
      } else if (act === "jump-date") {
        const ymd = actEl.dataset.date;
        const d = parseYMD(ymd);
        state.calYear = d.getFullYear();
        state.calMonth = d.getMonth();
        state.selectedDay = ymd;
        state.tab = "calendar";
        render();
      } else if (act === "set-profile-color") {
        state.profile.color = actEl.dataset.hex;
        saveLS("chulta4_profile", state.profile);
        render();
      } else if (act === "save-profile") {
        const name = document.getElementById("set-name").value.trim();
        state.profile.name = name;
        rememberProfile(name, state.profile.color);
        saveLS("chulta4_profile", state.profile);
        showToast("내 프리셋이 저장되었습니다");
        render();
      } else if (act === "set-theme") {
        state.theme = actEl.dataset.val;
        saveLS("chulta4_theme", state.theme);
        render();
      } else if (act === "set-accent") {
        state.accent = actEl.dataset.val;
        saveLS("chulta4_accent", state.accent);
        render();
      } else if (act === "copy-link") {
        copyText(window.location.href, "공유 링크가 복사되었습니다");
      } else if (act === "copy-month-text") {
        copyText(buildMonthSummaryText(), "월간 일정 요약이 복사되었습니다");
      } else if (act === "switch-room") {
        const val = document.getElementById("room-input").value.trim().replace(/[^a-zA-Z0-9_-]/g, "");
        if (!val) { showToast("영문 또는 숫자 룸 코드를 입력해 주세요"); return; }
        connectRoom(val);
        showToast(`#${val} 룸으로 이동했습니다`);
      } else if (act === "random-room") {
        const code = "r" + Math.random().toString(36).slice(2, 8);
        connectRoom(code);
        showToast(`새 공유 룸 #${code} 생성됨`);
      } else if (act === "export-json") {
        const blob = new Blob([JSON.stringify({ roomId: state.roomId, exportedAt: new Date().toISOString(), events: state.events, logs: state.logs }, null, 2)], { type: "application/json" });
        const a = document.createElement("a");
        a.href = URL.createObjectURL(blob);
        a.download = `chulta4-${state.roomId}-${todayYMD()}.json`;
        a.click();
        showToast("백업 파일을 다운로드했습니다");
      }
    });

    document.getElementById("btn-share").onclick = () => copyText(window.location.href, "공유 링크가 복사되었습니다");
    document.getElementById("btn-theme-toggle").onclick = () => {
      state.theme = state.theme === "dark" ? "light" : "dark";
      saveLS("chulta4_theme", state.theme);

    };
    document.getElementById("fab-add").onclick = () => openEventFormSheet({ startDate: state.selectedDay, endDate: state.selectedDay });
    document.getElementById("sheet-close").onclick = closeSheet;

      if (e.target.id === "sheet-backdrop") closeSheet();
    };
